function showSeats(tripKey) { document.getElementById('detailSection').style.display = 'none'; const data = tripsData[tripKey]; if (!data) return; document.getElementById('seatTripTitle').innerText = '🚌 ผังที่นั่ง: ' + data.title; document.getElementById('seatTripDate').innerText = data.date; const container = document.getElementById('vanWrapper'); container.innerHTML = ''; data.vans.forEach(van => { let seatsHtml = ''; van.seats.forEach(s => { let seatClass = 'seat'; if (s.num === 'driver') seatClass += ' driver'; else if (s.status === 'ว่าง') seatClass += ' available'; else if (s.status === 'จองแล้ว') seatClass += ' booked'; seatsHtml += `
${s.num === 'driver' ? '' : 'ที่นั่ง ' + s.num} ${s.status}
`; }); container.innerHTML += `
${van.name}
${seatsHtml}
${van.quote}
`; }); document.getElementById('seatSection').style.display = 'block'; document.getElementById('seatSection').scrollIntoView({ behavior: 'smooth' }); }
